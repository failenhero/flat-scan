# flats-scan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and deploy a Docker container that polls the BGRied iframe target every 5 min, detects new rental listings in Gernsheim (postal code 64579), and emails them to a Proton inbox via Gmail SMTP.

**Architecture:** A single Node.js 22 process running as a long-running daemon inside one Docker container. `setInterval` triggers a `check()` cycle every 5 min: fetch HTML, parse with cheerio, filter by postal code, diff against persistent UUID state, send one email per new listing via Gmail SMTP. State is a JSON file on a host-bound volume; a heartbeat file feeds Docker's HEALTHCHECK. Modules are dependency-injected so unit and integration tests can substitute fakes for network and SMTP.

**Tech Stack:** Node.js 22 LTS (alpine), `cheerio` (HTML parsing), `nodemailer` (SMTP), built-in `node:test` runner, built-in `fetch`. Docker + docker-compose for deployment.

**Spec:** [`docs/superpowers/specs/2026-05-09-flats-scan-design.md`](../specs/2026-05-09-flats-scan-design.md)

```javascript
import * as cheerio from 'cheerio';

const UUID_RE = /([0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12})/i;

export function parse(html) {
  const $ = cheerio.load(html);
  const container = $('div.properties').first();
  if (container.length === 0) return null;

  const listings = [];
  container.find('div.property').each((_, el) => {
    const $el = $(el);
    const $a = $el.find('h3.property-title a').first();
    const url = $a.attr('href');
    const title = $a.text().trim();
    if (!url || !title) return;

    const m = url.match(UUID_RE);
    if (!m) return;
    const uuid = m[1].toLowerCase();

    const address = $el.find('.property-subtitle').first().text().trim();
    const thumbnailUrl = $el.find('a.thumbnail img').first().attr('src') ?? '';

    listings.push({ uuid, title, address, url, thumbnailUrl });
  });

  return { listings };
}
```

```javascript
import { promises as fs } from 'node:fs';
import { dirname } from 'node:path';

export async function load(path) {
  let text;
  try {
    text = await fs.readFile(path, 'utf8');
  } catch (e) {
    if (e.code === 'ENOENT') {
      return { seen: new Set(), isFirstRun: true };
    }
    throw e;
  }
  const parsed = JSON.parse(text);
  return { seen: new Set(parsed.seenUuids ?? []), isFirstRun: false };
}

export async function save(path, seen) {
  await fs.mkdir(dirname(path), { recursive: true });
  const tmp = `${path}.tmp.${process.pid}.${Date.now()}`;
  const data = JSON.stringify({ seenUuids: [...seen] }, null, 2);
  await fs.writeFile(tmp, data, 'utf8');
  await fs.rename(tmp, path);
}
```

```javascript
const DEFAULT_TIMEOUT_MS = 30_000;
const DEFAULT_RETRY_DELAY_MS = 2_000;

export async function fetchHtml(url, opts = {}) {
  const {
    contactEmail = '',
    timeoutMs = DEFAULT_TIMEOUT_MS,
    retryDelayMs = DEFAULT_RETRY_DELAY_MS,
  } = opts;
  const userAgent = `flats-scan/1.0 (+${contactEmail})`;
  const headers = { 'User-Agent': userAgent, 'Accept-Language': 'de' };

  let lastError;
  for (let attempt = 1; attempt <= 2; attempt++) {
    try {
      const res = await fetch(url, {
        headers,
        signal: AbortSignal.timeout(timeoutMs),
      });
      if (res.status >= 500 && res.status < 600) {
        lastError = new Error(`HTTP ${res.status}`);
        if (attempt === 1) {
          await sleep(retryDelayMs);
          continue;
        }
        throw lastError;
      }
      if (!res.ok) {
        throw new Error(`HTTP ${res.status}`);
      }
      return await res.text();
    } catch (err) {
      lastError = err;
      const isLast = attempt === 2;
      const isHttpError = /^HTTP \d+$/.test(err.message);
      if (isLast || isHttpError) throw err;
      await sleep(retryDelayMs);
    }
  }
  throw lastError;
}

function sleep(ms) {
  return new Promise((r) => setTimeout(r, ms));
}
```

```javascript
import nodemailer from 'nodemailer';

export function buildTransporter({ host, port, user, pass }) {
  return nodemailer.createTransport({
    host,
    port,
    secure: port === 465,
    auth: { user, pass },
  });
}

export function buildEmail(listing) {
  const subject = `Neue Gernsheim-Wohnung: ${listing.title}`;
  const t = escapeHtml(listing.title);
  const a = escapeHtml(listing.address);
  const html =
    `<p>Eine neue Wohnung in Gernsheim ist verfügbar:</p>\n` +
    `<p><strong>${t}</strong></p>\n` +
    `<p>${a}</p>\n` +
    `<p><a href="${listing.url}">Zum Inserat</a></p>\n` +
    `<p><img src="${listing.thumbnailUrl}" alt="${t}" style="max-width:480px"></p>`;
  const text =
    `Eine neue Wohnung in Gernsheim ist verfügbar:\n\n` +
    `${listing.title}\n${listing.address}\n${listing.url}\n`;
  return { subject, html, text };
}

export async function send(transporter, message) {
  await transporter.sendMail(message);
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"]/g, (c) =>
    ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' })[c],
  );
}
```

```javascript
    return;

    .sort((a, b) => a.uuid.localeCompare(b.uuid));
  log.log(`diff new=${newOnes.length}`);

        await mailerFn(l);
        seen.add(l.uuid);
        await saveStateFn(config.statePath, seen);
        log.log(`mail sent uuid=${l.uuid}`);
      } catch (err) {
        log.error(`mail error uuid=${l.uuid}: ${err.message}`);
      }
    }
  }

  await heartbeatFn(config.heartbeatPath);
  log.log('tick ok');
}

export async function main() {

      to: config.mail.to,
```
